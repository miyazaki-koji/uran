MainWindow.xamlのコードを記述したViewModelであるMainViewModel.csは下記。


ViewModel 
using System;

// ObservableCollection
using System.Collections.Generic;

using System.Linq;
using System.Windows;

//INotifyPropertyChanged
//PropertyChanged
using System.ComponentModel;

//参照設定が必要
//using System.Configuration;


using Livet;
using Livet.Commands;
using Livet.Messaging;
//using Livet.Messaging.IO;
//using Livet.EventListeners;

//CloseCommand
using Livet.Messaging.Windows;

using System.Data.OracleClient;

//ICommand
using System.Windows.Input;

using LivetWPFApplication100.Models;

using Common;

namespace LivetWPFApplication100.ViewModels
{
    public class MainViewModel : ViewModel
    {
        #region 変更通知プロパティ
        //-----------------------------------------------
        private string _txtMessage;
        public string txtMessage
        {
            get { return _txtMessage; }
            set
            {
                _txtMessage = value;
                RaisePropertyChanged("txtMessage");
            }
        }
        //-----------------------------------------------
        private string _txt本日;
        public string txt本日
        {
            get { return _txt本日; }
            set
            {
                _txt本日 = value;
                RaisePropertyChanged("txt本日");
            }
        }
        //-----------------------------------------------
        private string _txtユーザーID;
        public string txtユーザーID
        {
            get { return this._txtユーザーID; }
            set
            {
                if (this.txtユーザーID == value)
                    return;

                this._txtユーザーID = value;
                RaisePropertyChanged("txtユーザーID");
            }
        }
        private string _txtパスワード;
        public string txtパスワード
        {
            get { return this._txtパスワード; }
            set
            {
                if (this.txtパスワード == value)
                    return;

                this._txtパスワード = value;
                RaisePropertyChanged("txtパスワード");
            }
        }
        #endregion

        Window win = null;
        public MainViewModel()
        {
            DB.str本日 = Convert.ToString(DateTime.Today.ToShortDateString());
            txt本日 = DB.str本日;
            Loaded = new Livet.Commands.ListenerCommand<Window>((w) =>
            {
                if (NeedHideOwner && w.Owner != null && 
                		w.Owner.Visibility == Visibility.Visible)
                {
                    win = w;
                    //w.Owner.Hide();
                }
            });
            
            Closing = new Livet.Commands.ListenerCommand<Window>((w) =>
            {
                if (NeedHideOwner && w.Owner != null)
                {
                    w.Owner.Show();
                }
            });
        }

        public bool NeedHideOwner { get; set; }
        public ICommand Loaded { get; private set; }
        public ICommand Closing { get; private set; }

        public void Initialize()
        {
        	if (win != null) win.Owner.Hide();
            var window = Application.Current.Windows.OfType<Window>().
            　　　　　　　　　　　　　　　　　　SingleOrDefault((w) => w.IsActive);
            　　　　　　　　　　　　　　　　　　XCloseButtonManager.Disable(window);

        }
        //------------------------------------------------------------------------
        public ICommand GotoCommand21
        {
            get { return new Livet.Commands.ViewModelCommand(Goto21); }
        }
        public void Goto21()
        {
            var window = Application.Current.Windows.OfType<Window>().
            　　　　　　　　　　　　　SingleOrDefault((w) => w.IsActive);
            window.Hide();
            Messenger.Raise(new TransitionMessage(new ViewModel2(), "MessageKey2"));
            window.Show();
        }
        public ICommand GotoCommand22
        {
            get { return new Livet.Commands.ViewModelCommand(Goto22); }
        }
        public void Goto22()
        {
            if (txtユーザーID == null)
            {
                MessageBox.Show("ユーザーIDが未入力です");
                return;
            }
            if (txtパスワード == null)
            {
                MessageBox.Show("パスワードが未入力です");
                return;
            }            
            if (!作業員マスターログインチェック(txtユーザーID.ToUpper(), 
            　　　　　　　　　　　　　　　　　　　　　　txtパスワード))
            {
                MessageBox.Show("ユーザーIDが違うかパスワードが違います");
                return;
            }
            
            Messenger.Raise(new TransitionMessage(new ViewModel2() { NeedHideOwner 
            = true }, "MessageKey2"));
        }
        //-----------------------------------------------------------------------
        public ICommand GotoCommand23
        {
            get { return new Livet.Commands.ViewModelCommand(Goto23); }
        }
        public void Goto23()
        {
            Messenger.Raise(new TransitionMessage(new ViewModel2() { NeedHideOwner 
            = true }, "MessageKey2"));
        }
        //-----------------------------------------------------------------------

        #region CloseCommand
        private ViewModelCommand _CloseCommand;
        public ViewModelCommand CloseCommand
        {
            get
            {
                if (_CloseCommand == null)
                {
                    _CloseCommand = new ViewModelCommand(Close);
                }
                return _CloseCommand;
            }
        }
        public void Close()
        {
            //Messenger.Raise(new WindowActionMessage("Close", WindowAction.Close));
            var window = Application.Current.Windows.OfType<Window>().
            　　　　　　　　　　　　　　　　　SingleOrDefault((w) => w.IsActive);
            window.Close();
        }
        #endregion

        #region 作業員マスターログインチェック
        public bool 作業員マスターログインチェック(string str作業員コード, 
        　　　　　　　　　　　　　　　　　　　　　　　　　　　　　string strパスワード)
        {
            bool bRet = false;

            using (OracleConnection con = new OracleConnection())
            {

                con.ConnectionString = "User Id=PRJDBA; Password=PRJ; 
                Data Source=NID08;Pooling=false;";
                con.Open();
                //

                string strSQL = String.Format("SELECT COUNT(*) AS CNT FROM 
                作業員マスター Where 作業員コード = '{0}' AND パスワード = '{1}'", 
                str作業員コード.ToUpper(), strパスワード);

                OracleCommand oracmd = new OracleCommand(strSQL, con);
                OracleDataReader reader = oracmd.ExecuteReader();

                //*** Tran ***
                //tx.Commit();
                while (reader.Read() == true)
                {
                    if (Convert.ToInt16(reader["CNT"].ToString()) > 0)
                    {
                        bRet = true;
                    }
                }
                return bRet;
            }
        }
        #endregion
    }
}
